public enum SimpleColor: Equatable {
  case red
  case green
  case blue

    nonisolated public static var generationSchema: FoundationModelsFake.GenerationSchema {
        FoundationModelsFake.GenerationSchema(type: Self.self, anyOf: ["red", "green", "blue"])
    }

    nonisolated public var generatedContent: FoundationModelsFake.GeneratedContent {
        switch self {
        case .red:
            "red".generatedContent
        case .green:
            "green".generatedContent
        case .blue:
            "blue".generatedContent
        }
    }
}

extension SimpleColor: nonisolated FoundationModelsFake.Generable {
    nonisolated public init(_ content: FoundationModelsFake.GeneratedContent) throws {
        let rawValue = try content.value(String.self)
        switch rawValue {
        case "red":
            self = .red
        case "green":
            self = .green
        case "blue":
            self = .blue
        default:
            throw FoundationModelsFake.GeneratedContent.ParsingError(rawContent: content.jsonString, debugDescription: "\"\(rawValue)\" isn't a case of \(Self.self)")
        }
    }
}
